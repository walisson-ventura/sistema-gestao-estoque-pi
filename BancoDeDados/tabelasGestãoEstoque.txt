-- 1. Tabela de Clientes
CREATE TABLE TB_CLIENTE (
    ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOME VARCHAR2(150) NOT NULL,
    CPF_CNPJ VARCHAR2(20) NOT NULL UNIQUE,
    TELEFONE VARCHAR2(20)
);

-- 2. Tabela de Fornecedores
CREATE TABLE TB_FORNECEDOR (
    ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOME VARCHAR2(150) NOT NULL,
    CNPJ VARCHAR2(20) NOT NULL UNIQUE,
    TELEFONE VARCHAR2(20)
);

-- 3. Tabela de Usuários (Vendedores/Operadores)
CREATE TABLE TB_USUARIO (
    ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOME VARCHAR2(150) NOT NULL,
    EMAIL VARCHAR2(150) NOT NULL UNIQUE,
    SENHA VARCHAR2(255) NOT NULL
);

-- 4. Tabela de Produtos
CREATE TABLE TB_PRODUTO (
    ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    NOME VARCHAR2(150) NOT NULL,
    DESCRICAO VARCHAR2(4000),
    PRECO NUMBER(10,2) NOT NULL,
    QUANTIDADE_ESTOQUE NUMBER NOT NULL,
    FORNECEDOR_ID NUMBER,
    CUSTO NUMBER(10,2),
    CONSTRAINT FK_PRODUTO_FORNECEDOR FOREIGN KEY (FORNECEDOR_ID) 
        REFERENCES TB_FORNECEDOR(ID)
);

-- 5. Tabela de Vendas
CREATE TABLE TB_VENDA (
    ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    CLIENTE_ID NUMBER NOT NULL,
    USUARIO_ID NUMBER NOT NULL,
    DATA_VENDA TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    VALOR_TOTAL NUMBER(10,2) NOT NULL,
    CONSTRAINT FK_VENDA_CLIENTE FOREIGN KEY (CLIENTE_ID) 
        REFERENCES TB_CLIENTE(ID),
    CONSTRAINT FK_VENDA_USUARIO FOREIGN KEY (USUARIO_ID) 
        REFERENCES TB_USUARIO(ID)
);

-- 6. Tabela de Itens da Venda
CREATE TABLE TB_ITEM_VENDA (
    ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    VENDA_ID NUMBER NOT NULL,
    PRODUTO_ID NUMBER NOT NULL,
    QUANTIDADE NUMBER NOT NULL,
    PRECO_UNITARIO NUMBER(10,2) NOT NULL,
    CONSTRAINT FK_ITEM_VENDA FOREIGN KEY (VENDA_ID) 
        REFERENCES TB_VENDA(ID) ON DELETE CASCADE,
    CONSTRAINT FK_ITEM_PRODUTO FOREIGN KEY (PRODUTO_ID) 
        REFERENCES TB_PRODUTO(ID)
);
